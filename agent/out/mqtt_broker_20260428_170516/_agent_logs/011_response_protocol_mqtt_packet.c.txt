#include "protocol/mqtt_packet.h"
#include <stdlib.h>

void mqtt_packet_free(mqtt_packet_t* p) {
    if (p == NULL) {
        return;
    }

    switch (p->type) {
        case MQTT_PKT_CONNECT: {
            mqtt_connect_payload_t* connect = &p->v.connect;
            if (connect->client_id != NULL) {
                free(connect->client_id);
                connect->client_id = NULL;
            }
            break;
        }
        case MQTT_PKT_PUBLISH: {
            mqtt_publish_payload_t* publish = &p->v.publish;
            if (publish->topic_name != NULL) {
                free(publish->topic_name);
                publish->topic_name = NULL;
            }
            if (publish->payload != NULL) {
                free(publish->payload);
                publish->payload = NULL;
            }
            publish->payload_len = 0;
            break;
        }
        case MQTT_PKT_SUBSCRIBE: {
            mqtt_subscribe_payload_t* subscribe = &p->v.subscribe;
            if (subscribe->topics != NULL) {
                for (size_t i = 0; i < subscribe->topic_count; ++i) {
                    if (subscribe->topics[i].filter != NULL) {
                        free(subscribe->topics[i].filter);
                        subscribe->topics[i].filter = NULL;
                    }
                }
                free(subscribe->topics);
                subscribe->topics = NULL;
            }
            subscribe->topic_count = 0;
            break;
        }
        default:
            break;
    }

    p->type = MQTT_PKT_RESERVED;
}
